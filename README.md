## 🗄️ 3. Database Schema & SQL 명령어

게시판 및 댓글의 다중 파일 첨부, 미디어 연동을 위해 Supabase `SQL Editor`에서 아래 쿼리를 실행하여 스키마를 구성합니다.

```sql
-- 1. posts 테이블 컬럼 추가 (다중 파일 및 기존 이미지 연동용)
ALTER TABLE posts ADD COLUMN IF NOT EXISTS attachments jsonb;
ALTER TABLE posts ADD COLUMN IF NOT EXISTS image_url text;

-- 2. comments 테이블 구성 (댓글 및 댓글 첨부파일용)
CREATE TABLE IF NOT EXISTS comments (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  post_id bigint REFERENCES posts(id) ON DELETE CASCADE,
  user_email text NOT NULL,
  content text,
  image_url text,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);
